function collectBindings(node: ts.Node): void {

function collectObject(node: ts.Node, prefix: string): void {
  if (ts.isPropertyAccessExpression(node) && ts.isIdentifier(node.expression)) {
    let object = bindings.get(node.expression.text);
    if (object && ts.isSatisfiesExpression(object)) object = object.expression;
    if (object && ts.isObjectLiteralExpression(object)) {
      const property = object.properties.find((field) => ts.isPropertyAssignment(field) && field.name.getText() === node.name.text);
      if (property && ts.isPropertyAssignment(property)) collectObject(property.initializer, prefix);
    }
    return;
  }
  if (ts.isIdentifier(node)) {
    const binding = bindings.get(node.text);
    if (binding) collectObject(binding, prefix);
    return;
  }
  if (ts.isSatisfiesExpression(node) || ts.isParenthesizedExpression(node)) return collectObject(node.expression, prefix);
  if (!ts.isObjectLiteralExpression(node)) return;
  for (const property of node.properties) {
    if (ts.isSpreadAssignment(property)) collectObject(property.expression, prefix);
    if (ts.isPropertyAssignment(property)) {
      const key = property.name.getText().replace(/^["']|["']$/g, "");
      const path = prefix + "." + key;
      const owner = property.getSourceFile();
      explicit.set(path, {value: property.initializer.getText(), location: `${owner.fileName}:${owner.getLineAndCharacterOfPosition(property.getStart()).line + 1}`});
      collectObject(property.initializer, path);
    } else if (ts.isMethodDeclaration(property)) {
      const owner = property.getSourceFile();
      explicit.set(prefix + "." + property.name.getText(), {value: property.getText(), location: `${owner.fileName}:${owner.getLineAndCharacterOfPosition(property.getStart()).line + 1}`});
    }
  }
}

  if (ts.isCallExpression(node) && node.expression.getText(source) === "adapter" && node.arguments[0]) collectObject(node.arguments[0], "StaticAdapter");
  if (ts.isCallExpression(node) && node.expression.getText(source) === "defineConfig") {
    const fn = node.arguments[0];

  }
  if (ts.isCallExpression(node) && node.expression.getText(source) === "sveltekit" && node.arguments[0]) collectObject(node.arguments[0], "SvelteKit");
  ts.forEachChild(node, scan);
}
